8th Program: Job Sequencing with Deadlines

#include <stdio.h>

int main()
{
    int n, i, j, temp;
    char job[20], tempJob;
    int deadline[20], profit[20];
    int timeslot[20];
    int maxDeadline = 0;
    int totalProfit = 0;

    printf("Enter number of jobs: ");
    scanf("%d", &n);

    printf("Enter Job ID, Deadline and Profit:\n");

    for (i = 0; i < n; i++)
    {
        scanf(" %c %d %d", &job[i], &deadline[i], &profit[i]);

        if (deadline[i] > maxDeadline)
            maxDeadline = deadline[i];
    }

    /* Sort jobs according to profit */
    for (i = 0; i < n - 1; i++)
    {
        for (j = i + 1; j < n; j++)
        {
            if (profit[i] < profit[j])
            {
                temp = profit[i];
                profit[i] = profit[j];
                profit[j] = temp;

                temp = deadline[i];
                deadline[i] = deadline[j];
                deadline[j] = temp;

                tempJob = job[i];
                job[i] = job[j];
                job[j] = tempJob;
            }
        }
    }

    /* Initialize time slots */
    for (i = 0; i < maxDeadline; i++)
    {
        timeslot[i] = -1;
    }

    /* Schedule jobs */
    for (i = 0; i < n; i++)
    {
        for (j = deadline[i] - 1; j >= 0; j--)
        {
            if (timeslot[j] == -1)
            {
                timeslot[j] = i;
                totalProfit = totalProfit + profit[i];
                break;
            }
        }
    }

    printf("\nTime Slot\tJob\n");

    for (i = 0; i < maxDeadline; i++)
    {
        if (timeslot[i] != -1)
        {
            printf("%d\t\t%c\n", i + 1, job[timeslot[i]]);
        }
    }

    printf("\nMaximum Profit = %d\n", totalProfit);

    return 0;
}


9th Program: 0/1 Knapsack using Dynamic Programming

#include <stdio.h>

#define MAX 100

int max(int a, int b)
{
    return (a > b) ? a : b;
}

int main()
{
    int n, W;
    int w[MAX], p[MAX];
    int V[MAX][MAX];
    int i, j;

    printf("Enter number of items: ");
    scanf("%d", &n);

    printf("Enter capacity of knapsack: ");
    scanf("%d", &W);

    printf("Enter weights of items:\n");
    for (i = 1; i <= n; i++)
        scanf("%d", &w[i]);

    printf("Enter profits of items:\n");
    for (i = 1; i <= n; i++)
        scanf("%d", &p[i]);

    for (j = 0; j <= W; j++)
        V[0][j] = 0;

    for (i = 1; i <= n; i++)
    {
        for (j = 0; j <= W; j++)
        {
            if (w[i] > j)
            {
                V[i][j] = V[i - 1][j];
            }
            else
            {
                V[i][j] = max(V[i - 1][j],
                              V[i - 1][j - w[i]] + p[i]);
            }
        }
    }

    printf("\nMaximum Profit = %d\n", V[n][W]);

    printf("\nDP Table:\n");
    for (i = 0; i <= n; i++)
    {
        for (j = 0; j <= W; j++)
        {
            printf("%4d", V[i][j]);
        }
        printf("\n");
    }

    return 0;
}


10th Program: N-Queens Problem using Backtracking

#include <stdio.h>
#include <stdlib.h>

int n;
int x[20];

int place(int k, int i)
{
    int j;

    for (j = 1; j < k; j++)
    {
        if (x[j] == i || abs(x[j] - i) == abs(j - k))
            return 0;
    }

    return 1;
}

void nqueen(int k)
{
    int i, j;

    for (i = 1; i <= n; i++)
    {
        if (place(k, i))
        {
            x[k] = i;

            if (k == n)
            {
                printf("\nSolution:\n");

                for (i = 1; i <= n; i++)
                {
                    for (j = 1; j <= n; j++)
                    {
                        if (x[i] == j)
                            printf(" Q ");
                        else
                            printf(" . ");
                    }
                    printf("\n");
                }

                printf("The optimal solution is:");
                for (i = 1; i <= n; i++)
                    printf("%d ", x[i]);

                printf("\n");
            }
            else
            {
                nqueen(k + 1);
            }
        }
    }
}

int main()
{
    printf("Enter number of queens: ");
    scanf("%d", &n);

    nqueen(1);

    return 0;
}


11th Program: 0/1 Knapsack using Backtracking

#include <stdio.h>

int n, W;
int w[20], p[20];
int x[20], best[20];
int maxProfit = 0;

void knapsack(int i, int weight, int profit)
{
    int j;

    if (i == n)
    {
        if (profit > maxProfit)
        {
            maxProfit = profit;

            for (j = 0; j < n; j++)
                best[j] = x[j];
        }
        return;
    }

    if (weight + w[i] <= W)
    {
        x[i] = 1;
        knapsack(i + 1, weight + w[i], profit + p[i]);
    }

    x[i] = 0;
    knapsack(i + 1, weight, profit);
}

int main()
{
    int i, totalWeight = 0;

    printf("Enter number of items: ");
    scanf("%d", &n);

    printf("Enter capacity: ");
    scanf("%d", &W);

    printf("Enter weights:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &w[i]);

    printf("Enter profits:\n");
    for (i = 0; i < n; i++)
        scanf("%d", &p[i]);

    knapsack(0, 0, 0);

    printf("\nOptimal Solution: ");
    for (i = 0; i < n; i++)
    {
        printf("%d ", best[i]);

        if (best[i] == 1)
            totalWeight += w[i];
    }

    printf("\nSelected items: ");
    for (i = 0; i < n; i++)
    {
        if (best[i] == 1)
            printf("%d ", i + 1);
    }

    printf("\nMaximum Profit = %d\n", maxProfit);

    return 0;
}


12th Program: Travelling Salesman Problem using Branch and Bound

#include <stdio.h>

#define INF 9999

int n;
int cost[10][10];
int visited[10];
int path[10], bestPath[10];
int minCost = INF;

void tsp(int city, int count, int totalCost)
{
    int i;

    if (count == n)
    {
        if (cost[city][0] != 0 &&
            totalCost + cost[city][0] < minCost)
        {
            minCost = totalCost + cost[city][0];

            for (i = 0; i < n; i++)
                bestPath[i] = path[i];
        }
        return;
    }

    for (i = 0; i < n; i++)
    {
        if (!visited[i] && cost[city][i] != 0)
        {
            if (totalCost + cost[city][i] < minCost)
            {
                visited[i] = 1;
                path[count] = i;

                tsp(i, count + 1,
                    totalCost + cost[city][i]);

                visited[i] = 0;
            }
        }
    }
}

int main()
{
    int i, j;

    printf("Enter number of cities: ");
    scanf("%d", &n);

    printf("Enter cost matrix:\n");

    for (i = 0; i < n; i++)
        for (j = 0; j < n; j++)
            scanf("%d", &cost[i][j]);

    for (i = 0; i < n; i++)
        visited[i] = 0;

    visited[0] = 1;
    path[0] = 0;

    tsp(0, 1, 0);

    printf("\nMinimum Cost = %d\n", minCost);

    printf("Optimal Path: ");

    for (i = 0; i < n; i++)
        printf("%d -> ", bestPath[i] + 1);

    printf("1\n");

    return 0;
}
